Format Response API (BERLAKU UNTUK SEMUA ENDPOINT):
  Sukses:
    { success: true, message: "Pesan sukses", data: { ... } }
  Error:
    { success: false, message: "Pesan error", errors?: { field: "pesan" } }
  Paginasi:
    { success: true, message: "...", data: [...], meta: { page, limit, total, totalPages } }

Autentikasi: Bearer Token di Header
  Header: Authorization: Bearer <accessToken>
  Simpan token di localStorage key: 'accessToken' dan 'refreshToken'
  401 = token expired → interceptor di axios.ts otomatis panggil refresh
  429 = rate limited → tampilkan pesan "Terlalu banyak percobaan"

Cara Akses Data di Response:
  const res = await api.get('/auth/me');
  const userData = res.data.data;     // ← ini data yg kamu butuhkan
  const message = res.data.message;   // ← pesan dari backend

    ```tsx
    import { useForm } from 'react-hook-form';
    import { zodResolver } from '@hookform/resolvers/zod';
    import { z } from 'zod';
    import { useAuth } from '../../contexts/AuthContext';
    import toast from 'react-hot-toast';

    const loginSchema = z.object({
      email: z.string().min(1, 'Email wajib diisi').email('Format email salah'),
      password: z.string().min(1, 'Password wajib diisi'),
    });
    type LoginForm = z.infer<typeof loginSchema>;

    export default function LoginPage() {
      const { login } = useAuth();
      const [serverError, setServerError] = useState('');
      const { register, handleSubmit, formState: { errors, isSubmitting } } = useForm<LoginForm>({
        resolver: zodResolver(loginSchema),
      });

      async function onSubmit(data: LoginForm) {
        try {
          setServerError('');
          await login(data.email, data.password);
          // redirect otomatis via PublicRoute
        } catch (err: any) {
          const msg = err.response?.data?.message || 'Terjadi kesalahan';
          if (err.response?.status === 429) {
            toast.error('Terlalu banyak percobaan. Tunggu beberapa menit.');
          } else {
            setServerError(msg);
          }
        }
      }

      return (
        <form onSubmit={handleSubmit(onSubmit)}>
          {/* ... form fields ... */}
          {serverError && <p className="text-red-500">{serverError}</p>}
          <button type="submit" disabled={isSubmitting}>
            {isSubmitting ? 'Memproses...' : 'Masuk'}
          </button>
        </form>
      );
    }
    ```

[ ] 2.2 - Buat Custom Hook: src/hooks/useMasterData.ts
    Hook ini memfetch & cache data master agar tidak dipanggil berulang.

    ```ts
    import { useState, useEffect } from 'react';
    import { masterApi } from '../api/master';

    export function useMasterData(type: 'kategori'|'pendidikan'|'instansi') {
      const [data, setData] = useState<any[]>([]);
      const [loading, setLoading] = useState(true);

      useEffect(() => {
        const fetcher = {
          kategori: masterApi.getKategori,
          pendidikan: masterApi.getPendidikan,
          instansi: masterApi.getInstansi,
        }[type];

        fetcher()
          .then(res => setData(res.data.data))
          .finally(() => setLoading(false));
      }, [type]);

      return { data, loading };
    }
    ```
